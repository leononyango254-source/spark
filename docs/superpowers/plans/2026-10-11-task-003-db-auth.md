> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Implement the Phase 1 foundation — Postgres schema (12 tables), Drizzle migrations, typed models/repositories, and the opaque-session authentication layer (POST `/api/auth/telegram`, GET `/api/me`) driven by validated Telegram `initData`. No blockchain money movement.

**Architecture:** `@spark/db` owns the Drizzle schema, migrations, client factory, and repositories. `@spark/domain` gains deterministic public-id generation. `@spark/rewards` and `@spark/ledger` state/type sets are reconciled to PRD §18/§21/§24. `apps/api` wires Fastify routes + an auth resolver on top of `@spark/db` and `@spark/telegram`. Integration tests provision an ephemeral Postgres 17 via Testcontainers and run the real migrations.

**Tech Stack:** PostgreSQL 17, Drizzle ORM 0.44 + drizzle-kit 0.31, node-postgres (`pg`), Testcontainers (`@testcontainers/postgresql`), Fastify 5, Zod 4, Vitest 3, pnpm 12.

## Conventions & guardrails (PRD §11, §19, §26, §53, §80)

- All money is integer micro-USDC stored as Postgres `bigint` (`{ mode: "bigint" }`); JS values are `bigint`. Never float.
- Identity/authorization fields are server-derived. The client sends only `initData` and task ids.
- Tokens are opaque and stored **hashed** (SHA-256). Raw tokens never persisted.
- Legacy `Spark prd.md` is gone; paths below are relative to repo root.
- Commands assume `~/.local/bin` is on `PATH` (pnpm lives there). If not, prefix with `export PATH="$HOME/.local/bin:$PATH"`.

```ts
import { randomBytes } from "node:crypto";

export const PUBLIC_ID_PATTERN = /^[0-9a-z]{16,32}$/;

/** URL-safe campaign public id. 12 random bytes -> 16 base32-ish lowercase chars. */
export function generatePublicId(): string {
  const alphabet = "0123456789abcdefghijklmnopqrstuvwxyz";
  const bytes = randomBytes(16);
  let out = "";
  for (const byte of bytes) {
    out += alphabet[byte & 31];
  }
  return out;
}
```

```ts

```

Note: `randomBytes(16)` with `& 31` yields 16 chars in the `a–z0–9` alphabet, matching `PUBLIC_ID_PATTERN` (`{16,32}`).

```ts
import { PostgreSqlContainer, type StartedPostgreSqlContainer } from "@testcontainers/postgresql";
import { drizzle } from "drizzle-orm/node-postgres";
import { migrate } from "drizzle-orm/node-postgres/migrator";
import { Pool } from "pg";
import * as schema from "../src/schema";

export interface Harness {
  container: StartedPostgreSqlContainer;
  pool: Pool;
  db: ReturnType<typeof drizzle<typeof schema>>;
  stop(): Promise<void>;
}

export async function startHarness(): Promise<Harness> {
  const container = await new PostgreSqlContainer("postgres:17").start();
  const pool = new Pool({ connectionString: container.getConnectionUri() });
  const db = drizzle(pool, { schema });
  await migrate(db, { migrationsFolder: new URL("../drizzle", import.meta.url).pathname });
  return {
    container,
    pool,
    db,
    async stop() {
      await pool.end();
      await container.stop();
    },
  };
}
```

```ts
import { createSession, upsertUserByTelegramId, type Db, type UserRow } from "@spark/db";
import { validateInitData } from "@spark/telegram";
import { generateSessionToken } from "./token";

export interface AuthResult {
  token: string;
  expiresAt: Date;
  user: UserRow;
}

export type AuthOutcome =
  | { ok: true; result: AuthResult }
  | { ok: false; reason: string };

export async function authenticateTelegram(
  db: Db,
  input: { initData: string; botToken: string; sessionTtlSeconds: number; now?: number },
): Promise<AuthOutcome> {
  const validation = validateInitData(input.initData, input.botToken, {
    now: input.now,
  });
  if (!validation.ok) return { ok: false, reason: validation.reason };
  if (!validation.user) return { ok: false, reason: "missing_user" };

  const user = await upsertUserByTelegramId(db, {
    telegramUserId: BigInt(validation.user.id),
    username: validation.user.username ?? null,
    firstName: validation.user.first_name ?? null,
    lastName: validation.user.last_name ?? null,
    languageCode: validation.user.language_code ?? null,
  });

  const { token, tokenHash } = generateSessionToken();
  const expiresAt = new Date((input.now ?? Math.floor(Date.now() / 1000)) * 1000 + input.sessionTtlSeconds * 1000);
  await createSession(db, { userId: user.id, tokenHash, expiresAt });

  return { ok: true, result: { token, expiresAt, user } };
}
```

```ts
import { findActiveSessionByTokenHash, findUserById, type Db, type UserRow } from "@spark/db";
import { hashToken } from "./token";

export function extractToken(headers: {
  authorization?: string | string[];
  cookie?: string;
}): string | null {
  const auth = headers.authorization;
  const bearer = Array.isArray(auth) ? auth[0] : auth;
  if (bearer?.toLowerCase().startsWith("bearer ")) return bearer.slice(7).trim();
  const cookie = headers.cookie;
  if (cookie) {
    for (const part of cookie.split(";")) {
      const [k, ...v] = part.trim().split("=");
      if (k === "spark_session") return decodeURIComponent(v.join("="));
    }
  }
  return null;
}

export async function resolveSession(db: Db, token: string): Promise<UserRow | null> {
  const session = await findActiveSessionByTokenHash(db, hashToken(token));
  if (!session) return null;
  return findUserById(db, session.userId);
}
```

```ts
import type { FastifyInstance } from "fastify";
import { z } from "zod";
import { authenticateTelegram } from "../auth/authenticateTelegram";
import type { ApiDeps } from "../server";

const bodySchema = z.object({ initData: z.string().min(1) });

export function registerAuthRoutes(app: FastifyInstance, deps: ApiDeps): void {
  app.post("/api/auth/telegram", async (request, reply) => {
    if (!deps.telegramBotToken) {
      return reply.code(503).send({ error: "telegram_not_configured" });
    }
    const parsed = bodySchema.safeParse(request.body);
    if (!parsed.success) return reply.code(400).send({ error: "invalid_body" });

    const outcome = await authenticateTelegram(deps.db, {
      initData: parsed.data.initData,
      botToken: deps.telegramBotToken,
      sessionTtlSeconds: deps.sessionTtlSeconds,
    });
    if (!outcome.ok) return reply.code(401).send({ error: "unauthorized", reason: outcome.reason });

    const { token, expiresAt, user } = outcome.result;
    reply.setCookie("spark_session", token, {
      httpOnly: true,
      sameSite: "lax",
      secure: deps.nodeEnv === "production",
      path: "/",
      expires: expiresAt,
    });
    return reply.send({
      token,
      expiresAt: expiresAt.toISOString(),
      user: { id: user.id, telegramUserId: user.telegramUserId.toString(), username: user.username },
    });
  });
}
```

```ts
import { afterAll, beforeAll, describe, expect, it } from "vitest";

function freshInitData(now: number): string {
  return signInitData(
    {
      auth_date: String(now),
      query_id: "AAF",
      user: JSON.stringify({ id: 4242, first_name: "Alice", username: "alice" }),
    },
    BOT_TOKEN,
  );

describe("auth", () => {
  it("rejects a missing/invalid body", async () => {
    const app = buildServer(deps);
    const res = await app.inject({ method: "POST", url: "/api/auth/telegram", payload: {} });
    expect(res.statusCode).toBe(400);
    await app.close();
  });

  it("rejects tampered initData", async () => {
    const app = buildServer(deps);
    const bad = freshInitData(Math.floor(Date.now() / 1000)).replace("alice", "mallory");
    const res = await app.inject({
      method: "POST",
      url: "/api/auth/telegram",
      payload: { initData: bad },

    expect(res.statusCode).toBe(401);
    await app.close();

    const app = buildServer(deps);

      method: "POST",
      url: "/api/auth/telegram",
      payload: { initData: freshInitData(Math.floor(Date.now() / 1000)) },

      method: "GET",
      url: "/api/me",
      headers: { authorization: `Bearer ${body.token}` },
    });
    expect(me.statusCode).toBe(200);
    expect((me.json() as { username: string }).username).toBe("alice");
    await app.close();
  });

  it("rejects unauthenticated /api/me", async () => {
    const app = buildServer(deps);
    const res = await app.inject({ method: "GET", url: "/api/me" });
    expect(res.statusCode).toBe(401);
    await app.close();
  });
});
```
